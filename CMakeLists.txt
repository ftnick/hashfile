cmake_minimum_required(VERSION 3.20)
project(hashfile VERSION 0.0.0 LANGUAGES C)
add_executable(hashfile src/main.c)
target_compile_features(hashfile PRIVATE c_std_11)
if(MSVC)
    target_compile_options(hashfile PRIVATE /W4)
else()
    target_compile_options(hashfile PRIVATE -Wall -Wextra -pedantic)
endif()
